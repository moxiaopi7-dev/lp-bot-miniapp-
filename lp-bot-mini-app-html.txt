    <div id="result" class="card" style="display:none;">
        <h3>结果</h3>

    <script>
        const tg = window.Telegram.WebApp;
        tg.expand();
        tg.ready();

        const API = 'http://127.0.0.1:3000/api';
        let token = null;

        async function init() {
            try {
                const user = tg.initDataUnsafe.user;
                document.getElementById('status').textContent =
                    `✅ 已连接 - ${user.first_name || '用户' + user.id}`;

                // 认证
                const res = await fetch(`${API}/auth`, {
                    method: 'POST',
                    headers: {'Content-Type': 'application/json'},
                    body: JSON.stringify({ initData: tg.initData })
                });
                const data = await res.json();
                token = data.token;
            } catch(e) {
                document.getElementById('status').textContent = '❌ 连接失败: ' + e.message;
            }
        }

        async function api(path) {
            const res = await fetch(`${API}${path}`, {
                headers: {'Authorization': `Bearer ${token}`}
            });
            return res.json();
        }

        function show(data) {
            document.getElementById('result').style.display = 'block';
            document.getElementById('resultText').textContent = JSON.stringify(data, null, 2);
        }

        async function testConnection() {
            tg.showAlert('API: ' + API + '\nToken: ' + (token ? '✅' : '❌'));
        }

        async function showPositions() {
            try {
                const data = await api('/positions');
                show(data);
                tg.showAlert(`找到 ${data.length} 个仓位`);
            } catch(e) {
                tg.showAlert('错误: ' + e.message);
            }
        }

        async function showBalance() {
            try {
                const data = await api('/balance');
                show(data);
            } catch(e) {
                tg.showAlert('错误: ' + e.message);
            }
        }

        init();
    </script>
